cmake_minimum_required(VERSION 3.21)
project(AutoRetouchCpp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs photo objdetect)
add_executable(AutoRetouchCpp WIN32 src/main.cpp)
target_link_libraries(AutoRetouchCpp PRIVATE Qt6::Widgets ${OpenCV_LIBS})
target_include_directories(AutoRetouchCpp PRIVATE ${OpenCV_INCLUDE_DIRS})
if(MSVC)
  target_compile_options(AutoRetouchCpp PRIVATE /W4 /permissive-)
endif()
